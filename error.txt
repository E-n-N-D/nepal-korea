	function fn_resizeImg(obj,gbcd){
		var width = obj.width;
    	var height = obj.height;

    	if(gbcd == "02"){
    		//이미지 팝업 div 크기
    		var maxWidth = 500;
    		var maxHeight = 500;
    	}else{
    		//텍스트,이미지 팝업 div 크기
    		var maxWidth = 500;
    		var maxHeight = 400;
    	}

		//이미지 비율
		var basisRatio = maxHeight / maxWidth;
		var imgRatio = height / width;

		//팝업구분 : 이미지
		if(imgRatio > basisRatio){
			if(height > maxHeight){
				resizeHeight = maxHeight;
				resizeWidth = Math.round((width * resizeHeight) / height);
			}else{
				resizeWidth = width;
				resizeHeight = height;
			}
		}else if(imgRatio < basisRatio){
			if(width > maxWidth){
				resizeWidth = maxWidth;
				resizeHeight = Math.round((height * resizeWidth) / width);
			}else{
				resizeWidth = width;
				resizeHeight = height;
			}
		}else{
			resizeWidth = width;
			resizeHeight = height;
		}

		$("#"+obj.id).attr("style", "width:"+resizeWidth+"px; height:"+resizeHeight+"px; display: inline;");
    }

function get_version_of_IE() {
	var word;
	var version = "N/A";
	var agent = navigator.userAgent.toLowerCase();
	var name = navigator.appName;
	var vsCheck ='Y';

	// IE old version ( IE 10 or Lower )
	if ( name == "Microsoft Internet Explorer" ) word = "msie ";

	else {
		// IE 11
		if ( agent.search("trident") > -1 ) word = "trident/.*rv:";
		// Microsoft Edge
		else if ( agent.search("edge/") > -1 ) word = "edge/";
	}

	var reg = new RegExp( word + "([0-9]{1,})(\\.{0,}[0-9]{0,1})" );

	if (  reg.exec( agent ) != null  ) version = RegExp.$1 + RegExp.$2;

	return version;
}
